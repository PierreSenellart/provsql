\newcommand{\pvdoc}[7][black]{
  \begin{scope}[shift={(#3,#4)}]
    \pgfmathsetmacro{\pvw}{#5/2}\pgfmathsetmacro{\pvh}{#6/2}
    \path[pvline, draw=#1!70, fill=white]
      (-\pvw,-\pvh) -- (-\pvw,\pvh) -- ({\pvw-.2},\pvh) -- (\pvw,{\pvh-.2}) -- (\pvw,-\pvh) -- cycle;
    \path[pvline, draw=#1!70, fill=#1!18]
      ({\pvw-.2},\pvh) -- ({\pvw-.2},{\pvh-.2}) -- (\pvw,{\pvh-.2}) -- cycle;
    \node[pvtext, minimum width=#5 cm, minimum height=#6 cm] (#2) at (0,0) {#7};
  \end{scope}}
\newcommand{\pvcyl}[6]{
  \begin{scope}[shift={(#2,#3)}]
    \pgfmathsetmacro{\pvw}{#4/2}\pgfmathsetmacro{\pvh}{#5/2}\def\pvr{.13}
    \path[pvline, draw=isA, fill=isA!12]
      (-\pvw,{\pvh-\pvr}) -- (-\pvw,{-\pvh+\pvr})
      arc[start angle=180, end angle=360, x radius=\pvw, y radius=\pvr]
      -- (\pvw,{\pvh-\pvr})
      arc[start angle=0, end angle=180, x radius=\pvw, y radius=\pvr] -- cycle;
    \path[pvline, draw=isA, fill=isA!28] (0,{\pvh-\pvr}) ellipse[x radius=\pvw, y radius=\pvr];
    \node[minimum width=#4 cm, minimum height=#5 cm, inner sep=0pt] (#1) at (0,0) {};
    \node[pvtext] at (0,{-\pvr/1.4}) {#6};
  \end{scope}}
